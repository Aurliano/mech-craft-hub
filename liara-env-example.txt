# Liara Object Storage Configuration
# این فایل را در تنظیمات محیطی Liara اضافه کنید

# نوع storage
FILE_STORAGE_TYPE=liara

# نام باکت در لیارا
FILE_BUCKET_NAME=resources

# منطقه لیارا
FILE_REGION=iran

# دسترسی عمومی به فایل‌ها

# کلیدهای API لیارا (از پنل لیارا دریافت کنید)
LIARA_ACCESS_KEY=your-access-key
LIARA_SECRET_KEY=your-secret-key

# تنظیمات فایل
